## Solution Code

```c
#include <stdio.h>
#include <stdlib.h>

#define FOR_LOOP(i, n) for (int i = 0; i < n; i++)
#define IF_COND(c) if (c)

struct QNode {
    int key;
    struct QNode* next;
};

struct Queue {
    struct QNode *front, *rear;
};

struct QNode* newNode(int k) {
    struct QNode* temp = (struct QNode*)malloc(sizeof(struct QNode));
    temp->key = k;
    temp->next = NULL;
    return temp;
}

struct Queue* createQueue() {
    struct Queue* q = (struct Queue*)malloc(sizeof(struct Queue));
    q->front = q->rear = NULL;
    return q;
}

void enQueue(struct Queue* q,int k) {
    struct QNode* temp = newNode(k);
    IF_COND(q->rear == NULL) {
        q->front = q->rear = temp;
        return;
    }
    q->rear->next = temp;
    q->rear = temp;
}

void deQueue(struct Queue* q) {
    IF_COND(q->front == NULL) return;
    struct QNode* temp = q->front;
    q->front = q->front->next;
    IF_COND(q->front == NULL) q->rear = NULL;
    free(temp);
}

int main() {
    struct Queue* q = createQueue();
    int n, m, k;
    scanf("%d %d", &n, &m);
    FOR_LOOP(i, n) {
        scanf("%d", &k);
        enQueue(q, k);
    }
    printf("Front:%d\n", q->front->key);
    printf("Rear:%d\n", q->rear->key);
    printf("After 2 deQueue and M enqueue\n");
    deQueue(q);
    deQueue(q);
    FOR_LOOP(i, m) {
        scanf("%d", &k);
        enQueue(q, k);
    }
    printf("Front is:%d\n", q->front->key);
    printf("Queue Rear:%d\n", q->rear->key);
    return 0;
}
```
